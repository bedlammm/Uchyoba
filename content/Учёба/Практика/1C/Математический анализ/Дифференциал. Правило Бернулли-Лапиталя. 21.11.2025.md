$$
\begin{gathered}
\begin{gathered}
dy=f'(x)\Delta x\\
dy=f'(x)dx
\end{gathered}\quad-\quad\text{Дифференциал функуции, линейная часть её приращения.}\\\\
dy=f'(x)dx\quad=>\quad f'(x)=\frac{dy}{dx}\\\\
\Delta x\quad-\text{\quadМало }=>\Delta y\approx dy\\
\Delta y=f(x+\Delta x)-f(x)\approx f'(x)\Delta x=dy\\
f(x+\Delta x)\approx f'(x)\Delta x+f(x)\quad -\quad\text{Первое приближение}
\end{gathered}
$$  
## $y=\cos x\quad x=\frac\pi6\quad\Delta x=\frac\pi{36}\quad\Delta y-?\quad dy-?$  
$$
\begin{gathered}
\Delta y=\cos(x+\Delta x)-\cos x\\
\Delta y=\cos(\frac\pi6+\frac\pi{36})-\cos \frac\pi6\\
\Delta y=\cos\frac{7\pi}{36}-\cos \frac\pi6\approx0.046\\
dy=-\sin x\Delta x\\
dy=-\sin\frac\pi6\cdot\frac\pi{36}\approx-0.043
\end{gathered}
$$  
## $\arcsin0.05\approx?$ 
$$
\begin{gathered}
y=\arcsin x\quad x=0\quad\Delta x=0.05\\
y'=\sqrt{\frac1{1-x^2}}\\
\arcsin(x+\Delta x)\approx\frac1{\sqrt{1-x^2}}\Delta x+\arcsin x\approx0.05
\end{gathered}
$$  
## $\arctan1.04\approx?$  
$$
\begin{gathered}
y=\arctan x\quad x=1\quad \Delta x=0.04\\
f(x+\Delta x) \approx f'(x) \cdot \Delta x + f(x) \\
f(1+0.04)\approx\frac1{1+1}\cdot0.04+\arctan1\approx0.02+\frac\pi4
\end{gathered}
$$  
## $\ln 1,2 \approx ?$  
$$
\begin{gathered}
x = 1 \quad \Delta x = 0,2\quad f(x) = \ln x \\
f(x+\Delta x) \approx f'(x) \cdot \Delta x + f(x) \\
\ln(1+0,2) \approx \frac{1}{1} \cdot 0,2 + \ln(1) \\
\ln(1,2) \approx 0,2
\end{gathered}
$$

# Правило Бернулли-Лапиталя
$$
\begin{gathered}
1) \ f(x) \text{ и } \varphi(x)\in D=> \forall x: a-h < x < a+h, \ \varphi'(x) \neq 0 \\
2) \ f(x) \text{ и } \varphi(x) - \text{б.б. (или б.м.)} \\
\text{Тогда, если } \exists \lim_{x \to a} \frac{f'(x)}{\varphi'(x)}=[\frac00]\text{ Или }=[\frac\infty\infty]\\
=>\lim_{x \to a} \frac{f(x)}{\varphi(x)}=\lim_{x \to a} \frac{f'(x)}{\varphi'(x)}
\end{gathered}
$$  
## $\lim_{x\to0}\frac{x\cos x-\sin x}{x^3}$  
$$
\begin{gathered}
\lim_{x \to 0} \frac{x \cos x - \sin x}{x^3} = \left[ \frac{0}{0} \right] \stackrel{\text{b-l}}{=} \\
= \lim_{x \to 0} \frac{1 \cdot \cos x + x \cdot (-\sin x) - \cos x}{3x^2} = \lim_{x \to 0} \frac{-x \sin x}{3x^2} = \\
= \lim_{x \to 0} \frac{-\sin x}{3x} = -\frac{1}{3}
\end{gathered}
$$  
## $\lim_{x\to0}\frac {e^{3x}-3x-1}{\sin^25x}$  
$$
\begin{gathered}
\lim_{x\to0}\frac {e^{3x}-3x-1}{\sin^25x}=[\frac00]=\lim_{x\to0}\frac {3e^{3x}-3}{5\sin10x}=[\frac00]=\lim_{x\to0}\frac {9e^{3x}}{50\cos10x}=[\frac{9}{50}]=0.18
\end{gathered}
$$  
## $\lim_{x\to+\infty}\frac{\ln x}{\sqrt[3]x}$  
$$
\lim_{x\to+\infty}\frac{\ln x}{\sqrt[3]x}=[\frac{+\infty}{+\infty}]=\lim_{x\to+\infty}-\frac{1}{3\sqrt[3]{x^7}}=[-\frac1\infty]=0
$$  
## $\lim_{x\to1}(1-x)\tan\frac{\pi x}{2}$ 
$$
\begin{gathered}
\lim_{x\to1}(1-x)\tan\frac{\pi x}{2}=[0\cdot\infty]=\lim_{x\to1}\frac{\tan\frac{\pi x}{2}}{\frac1{1-x}}=[\frac\infty\infty]=\lim_{x\to1}\frac{\frac\pi2(1-x)^2}{\cos^2\frac{\pi x}2}=[\frac00]=\\
=\lim_{x\to1}\frac{1-x}{\cos\frac{\pi x}2}=[\frac00]=\lim_{x\to1}\frac{-1}{-\sin(\frac{\pi x}{2})\frac\pi2}=\frac\pi2
\end{gathered}
$$  
## $\lim_{x\to1}(\ln x\cdot\ln(x-1))$  
$$
\begin{gathered}
\lim_{x\to1+0}(\ln x\cdot\ln(x-1))=[0\cdot(-\infty)]=^{б.л.}\lim_{x\to1+0}(\frac{\ln(x-1)}{\frac1{\ln x}})=[\frac{-\infty}0]\\
=-\lim_{x\to1+0}(\frac{x\ln x^2}{1-x})=[-\frac00]=^{б.л.}-\lim_{x\to1+0}(\frac{\frac1x2\ln x}{1})=0
\end{gathered}
$$
