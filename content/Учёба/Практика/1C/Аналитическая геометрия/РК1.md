## 2. Найти угол $\varphi$ между векторами $\vec{a} = \vec{m} + \vec{n}$ и $\vec{b} = \vec{m} - \vec{n}$, если $|\vec{m}| = \sqrt{2}$, $|\vec{n}| = 1$, $\angle(\vec{m}, \vec{n}) = \frac{\pi}{4}$.
$$
\begin{gathered}
\varphi=\arccos(\frac{\vec a\cdot\vec b}{|\vec a||\vec b|})\\
\varphi=\arccos(\frac{(\vec m+\vec n)\cdot(\vec m-\vec n)}{\sqrt{|\vec m+\vec n|^2}\sqrt{|\vec m-\vec n|^2}})\\
\varphi=\arccos(\frac{|\vec m|^2-|\vec n|^2}{\sqrt{||\vec m|^2+|\vec n|^2+2\vec m\vec n|}\sqrt{||\vec m|^2+|\vec n|^2-2\vec m\vec n|}})\\
\varphi=\arccos(\frac{2-1}{\sqrt{2+1+2\cdot1}\sqrt{2+1-2\cdot1}})\\
\varphi=\arccos(\frac{1}{\sqrt5})
\end{gathered}
$$
